    模块损毁时播放动画

    中文 | English

    把脚本挂在内部模块所在物体。将带有 Animator 的物体分配给 animator，并把 triggerName 设置为控制器中对应的 Trigger。示例使用 Destroyed。

    脚本在 OnVehicleReady 获取模块并订阅状态。模块首次进入损毁状态时触发动画，修复后可以再次触发。用 self.script 订阅，行为销毁时会解除订阅。

    Behavior()
Property = {
    animator = { type = "object", value = nil },
    triggerName = { type = "string", value = "Destroyed" },
}

local ModuleDestroyedTrigger = class("ModuleDestroyedTrigger")

function ModuleDestroyedTrigger:OnVehicleReady(vehicle)
    local module = InternalModuleAPI.GetInternalModule(self.gameObject)
    if module == nil or self.animator == nil then
        return
    end
    local animator = ComponentAPI.GetNativeComponent(self.animator, "Animator")
    if animator == nil then
        return
    end
    local wasDestroyed = InternalModuleAPI.IsDestroyed(module)
    InternalModuleAPI.OnStateChanged(self.script, module, function()
        local destroyed = InternalModuleAPI.IsDestroyed(module)
        if destroyed and not wasDestroyed then
            animator:SetTrigger(self.triggerName)
        end
        wasDestroyed = destroyed
    end)
end

return ModuleDestroyedTrigger
