    把脚本挂到载具，按需绑定 fan、smoke 或 gearLights。飞机先读取起落架初始状态，再用 FlightAPI.OnGearChanged 更新灯光。事件订阅传入 self.script，并关闭飞机分支的逐帧更新。

    地面分支从 TankAPI.GetEngineState 读取成功标记、转速、转速百分比、挡位、速度和倒车状态。每分钟转数乘 6 得到每秒角度，示例额外按 0.1 缩放风扇转速，并在转速百分比超过 80 时显示烟雾。

    Behavior()
Property = {
    fan = { type = "object", value = nil },
    smoke = { type = "object", value = nil },
    gearLights = { type = "object", value = nil },
}

local EngineAndGear = class("EngineAndGear")

function EngineAndGear:OnVehicleReady(vehicle)
    self.vehicle = vehicle
    if VehicleAPI.IsFlightVehicle(vehicle) then
        if self.gearLights ~= nil then
            self.gearLights:SetActive(FlightAPI.IsGearDown(vehicle))
            FlightAPI.OnGearChanged(self.script, vehicle, function(isDown)
                self.gearLights:SetActive(isDown)
            end)
        end
        self.vehicle = nil
        self.script:SetUpdateEnabled(false)
        return
    end
    self.script:SetUpdateEnabled(self.fan ~= nil or self.smoke ~= nil)
end

function EngineAndGear:OnUpdated(dt)
    if self.vehicle == nil then
        return
    end
    local ok, rpm, rpmPercent = TankAPI.GetEngineState(self.vehicle)
    if not ok then
        return
    end
    if self.fan ~= nil then
        self.fan.transform:Rotate(0, 0, rpm * 6 * dt * 0.1)
    end
    if self.smoke ~= nil then
        self.smoke:SetActive(rpmPercent > 80)
    end
end

return EngineAndGear
