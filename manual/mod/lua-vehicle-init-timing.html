    在 Lua API 2 及以上客户端，把需要载具部件的逻辑放到 OnVehicleReady。脚本先运行 OnStarted。如果载具已经装配完成，随后立即调用 OnVehicleReady，否则在装配完成后调用一次。

    将脚本绑定到载具上的 LuaBehaviorMono，用 InitVehicleDependencies 初始化自己的部件。兼容旧客户端时，先通过 TryGetTankInitSystem 获取载具，再检查 IsLoaded 或订阅加载事件。示例在行为销毁时解除旧版订阅。

    Behavior()
Property = {}

local VehicleTiming = class("VehicleTiming")

function VehicleTiming:OnVehicleReady(vehicle)
    if vehicle == nil or self.vehicle ~= nil then
        return
    end
    self.vehicle = vehicle
    self:InitVehicleDependencies()
end

function VehicleTiming:InitVehicleDependencies()
    print("[VehicleTiming] vehicle is ready")
end

function VehicleTiming:OnStarted()
    if (GAME_API_VERSION or 1) >= 2 then
        return
    end
    local found, vehicle = self.script:TryGetTankInitSystem()
    if not found or vehicle == nil then
        return
    end
    if vehicle.IsLoaded then
        self:OnVehicleReady(vehicle)
        return
    end
    self.legacyVehicle = vehicle
    self.legacyLoaded = function()
        self:OnVehicleReady(vehicle)
    end
    VehicleAPI.RegisterVehicleLoadedEvent(vehicle, self.legacyLoaded)
end

function VehicleTiming:OnDestroyed()
    if self.legacyLoaded ~= nil then
        VehicleAPI.UnRegisterVehicleLoaedEvent(self.legacyVehicle, self.legacyLoaded)
        self.legacyVehicle = nil
        self.legacyLoaded = nil
    end
end

return VehicleTiming
