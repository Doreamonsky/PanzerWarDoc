    Spawn random meteors in a scene with Lua

    中文 | English

    1. Add an empty GameObject to the scene and attach LuaBehaviorMono.
    2. Save the code below as RandomSpawn.lua, import it into Unity, and assign it to the component.
    3. Assign a meteor prefab to prefab in the Lua object properties. The example uses count, fieldSize, and noiseScale to control the object count, XZ range, and height variation. Start with a small number of objects.
    Behavior()
Property = {
    prefab = { type = "object", value = nil },
    count = { type = "number", value = 100 },
    fieldSize = { type = "number", value = 100 },
    noiseScale = { type = "number", value = 10 }
}

local RandomSpawn = class("RandomSpawn")
local Mathf = CS.UnityEngine.Mathf
local Vector3 = CS.UnityEngine.Vector3
local Quaternion = CS.UnityEngine.Quaternion

function RandomSpawn:OnStarted()
    if self.prefab == nil then
        return
    end

    local count = math.max(0, math.floor(self.count or 100))
    local fieldSize = math.max(0, self.fieldSize or 100)
    local noiseScale = math.max(0.01, self.noiseScale or 10)
    for i = 1, count do
        local x = math.random() * fieldSize
        local z = math.random() * fieldSize
        local y = Mathf.PerlinNoise(x / noiseScale, z / noiseScale) * fieldSize
        local meteor = CS.UnityEngine.Object.Instantiate(self.prefab, Vector3(x, y, z), Quaternion.identity)
        meteor.name = "Meteor_" .. i
    end
end

return RandomSpawn
