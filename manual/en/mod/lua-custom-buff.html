    Attach the script to the local player's vehicle. API 2 creates a mutable coefficient with BuffAPI.CreateValueBuff and adds it to the main weapon's ReloadTimeBuffedCoff. The example adds it with F1 and removes it with F2. With no other reload modifiers, -0.5 changes an eight-second reload to four seconds. Destroying the behaviour unregisters the keys and removes the attached buff.

    Behavior()
Property = {
    reloadCoefficient = { type = "number", value = -0.5 },
}

local ReloadBuffExample = class("ReloadBuffExample")

function ReloadBuffExample:OnVehicleReady(vehicle)
    if not VehicleAPI.IsLocalPlayer(vehicle) then
        return
    end
    local weapon = WeaponAPI.GetWeapon(vehicle, 0)
    if weapon == nil then
        return
    end
    self.stat = weapon.ReloadTimeBuffedCoff
    self.buff = BuffAPI.CreateValueBuff("Example reload time", false, self.reloadCoefficient)
    local id = tostring(self.gameObject:GetInstanceID())
    self.addAction = "ExampleReloadAdd_" .. id
    self.removeAction = "ExampleReloadRemove_" .. id
    InputAPI.RegisterKeyInput(self.addAction, "F1", function() self:Add() end, function() end)
    InputAPI.RegisterKeyInput(self.removeAction, "F2", function() self:Remove() end, function() end)
end

function ReloadBuffExample:Add()
    if self.buff ~= nil and not self.added then
        self.stat:AddBuff(self.buff)
        self.added = true
    end
end

function ReloadBuffExample:Remove()
    if self.added then
        self.stat:RemoveBuff(self.buff)
        self.added = false
    end
end

function ReloadBuffExample:OnDestroyed()
    if self.addAction ~= nil then
        InputAPI.UnregisterKeyInput(self.addAction)
        InputAPI.UnregisterKeyInput(self.removeAction)
    end
    self:Remove()
end

return ReloadBuffExample
