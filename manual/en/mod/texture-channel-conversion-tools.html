    Pack and unpack ORM textures

    中文 | English

    Run this command at the repository root to replay the examples through the local Unity MCP.

    python Docs/Manual/manual/tools/tutorials.py run texture-channel-conversion-tools
    Click to expand MCP actionsClick to collapse MCP actions

    Prepare the teaching sample. The recipe provides ${variables}.

    {
  "variables": {
    "texture": "${fixture}/TutorialORM.png"
  },
  "calls": [
    {
      "tool": "invoke",
      "arguments": {
        "path": "static:UnityEditor.AssetDatabase/CreateFolder",
        "args": [
          "${fixtureParent}",
          "${folder}"
        ]
      },
      "expect": {
        "contains": "(String)"
      }
    }
  ]
}
    1. In Substance 3D Designer, connect ambient occlusion, roughness, and metallic to R, G, and B on RGBA Merge. Connect a constant white value to A for an explicit opaque alpha. Adobe RGBA Merge
    2. To edit existing ORM, use RGBA Split, then reconnect the channels to RGBA Merge in the required order. Adobe RGBA Split
    3. Export a linear data texture and disable sRGB in Unity. See PBR mask channel merging for the Panzer War Mask mapping.

    The Unity sample writes three test values and reads back the PNG. Designer connections follow Adobe's documentation.

    [
  {
    "tool": "invoke",
    "arguments": {
      "path": "static:ShanghaiWindy.Core.AssetBundleManager/LuaEnv/DoString",
      "args": [
        "local U = CS.UnityEngine\nlocal source = U.Texture2D(2, 2, U.TextureFormat.RGBA32, false, true)\nlocal color = U.Color(0.18, 0.42, 0.72, 1)\nfor x = 0, 1 do\n    for y = 0, 1 do source:SetPixel(x, y, color) end\nend\nsource:Apply()\nlocal png = U.ImageConversion.EncodeToPNG(source)\nCS.System.IO.File.WriteAllBytes('${texture}', png)\nlocal copy = U.Texture2D(2, 2, U.TextureFormat.RGBA32, false, true)\nassert(U.ImageConversion.LoadImage(copy, png, false))\nlocal pixel = copy:GetPixel(0, 0)\nassert(math.abs(color.r - pixel.r) < .005 and math.abs(color.g - pixel.g) < .005)\nassert(math.abs(color.b - pixel.b) < .005 and math.abs(color.a - pixel.a) < .005)\nU.Object.DestroyImmediate(source)\nU.Object.DestroyImmediate(copy)\nreturn \"PNG channel sample verified\"\n",
        "tutorial-orm",
        "null"
      ]
    },
    "expect": {
      "contains": "PNG channel sample verified"
    }
  },
  {
    "tool": "invoke",
    "arguments": {
      "path": "static:UnityEditor.AssetDatabase/ImportAsset",
      "args": [
        "${texture}",
        "ForceSynchronousImport"
      ]
    },
    "expect": {
      "contains": "OK:"
    }
  },
  {
    "tool": "inspect_set",
    "arguments": {
      "path": "importer:${texture}/sRGBTexture",
      "value": "false"
    },
    "expect": {
      "contains": "OK:"
    }
  },
  {
    "tool": "inspect",
    "arguments": {
      "path": "importer:${texture}/sRGBTexture"
    },
    "expect": {
      "contains": "(Boolean) False"
    }
  }
]
