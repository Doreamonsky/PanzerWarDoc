    1. Configure the ground radar

    In VehicleTextData, expand groundAARadarData under AA Radar and enable hasRadar. Set the detection range, then adjust the search area and display ranges for the model.

    Field Meaning
    maxDetectionRange Maximum detection range in metres
    isFullRotation, fullRotationRate Full-rotation search and its rate in degrees per second
    searchAzimuthHalfAngle, searchElevationHalfAngle Azimuth and elevation half angles in degrees
    minElevationAngle, detectGroundTargets Ground clutter elevation filter and ground target detection
    displayRanges Selectable display ranges in metres
    canUsePD, pdMinDetectableSpeed Pulse-Doppler mode and its speed threshold
    hasIFF Friendly contact identification

    2. Bind the script and update the list

    Attach the script to the vehicle, obtain the radar in OnVehicleReady and subscribe to lock changes. The example refreshes the contact text every 0.2 seconds and draws the cached text in OnGUI. Read contacts with a zero-based index. A chaff return can have no vehicle. The list uses GetContactWorldAzimuth for world bearing.

    Behavior()
Property = {}

local RadarScreen = class("RadarScreen")
local GUI = CS.UnityEngine.GUI
local Rect = CS.UnityEngine.Rect

function RadarScreen:OnVehicleReady(vehicle)
    local ok, radar = RadarAPI.TryGetRadar(vehicle)
    if not ok then
        self.script:SetUpdateEnabled(false)
        return
    end
    self.vehicle = vehicle
    self.radar = radar
    self.lines = {}
    self.text = ""
    RadarAPI.OnLockChanged(self.script, radar, function(locked, target)
        if locked and target ~= nil then
            print("[Radar] locked " .. VehicleAPI.GetDisplayName(target))
        elseif not locked then
            print("[Radar] lock lost")
        end
    end)
    self.script:SetUpdateInterval(0.2)
end

function RadarScreen:OnUpdated()
    self.visible = self.radar ~= nil
        and VehicleAPI.IsLocalPlayer(self.vehicle)
        and RadarAPI.IsActive(self.radar)
    if not self.visible then
        return
    end
    local written = 0
    for index = 0, RadarAPI.GetContactCount(self.radar) - 1 do
        local ok, target, distance, azimuth, elevation, altitude, radialSpeed, isFriendly =
            RadarAPI.GetContact(self.radar, index)
        if ok then
            local name = target ~= nil and VehicleAPI.GetDisplayName(target) or "?"
            local bearing = RadarAPI.GetContactWorldAzimuth(self.radar, index)
            written = written + 1
            self.lines[written] = string.format("%s %s  %.1f km  %03.0f°  %.0f m",
                isFriendly and "+" or "!", name, distance / 1000, bearing, altitude)
        end
    end
    for index = written + 1, #self.lines do
        self.lines[index] = nil
    end
    self.text = table.concat(self.lines, "\n")
end

function RadarScreen:OnGUI()
    if self.visible then
        GUI.Label(Rect(20, 120, 360, 300), self.text)
    end
end

return RadarScreen
